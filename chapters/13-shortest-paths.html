13. Shortest paths

  • Bellman-Ford O(VE): chịu cạnh âm, phát hiện chu trình âm.
  • Dijkstra O((V+E) log V): không cạnh âm, dùng nhiều nhất.
  • Floyd-Warshall O(V³): mọi cặp đỉnh.
using pll=pair<long long,int>;
priority_queue<pll,vector<pll>,greater<pll>> pq;
vector<long long> d(n+1,LLONG_MAX);
d[s]=0; pq.push({0,s});
while(!pq.empty()){
  auto [du,u]=pq.top(); pq.pop();
  if(du!=d[u]) continue;
  for(auto [v,w]:g[u]) if(d[v]>d[u]+w){
    d[v]=d[u]+w; pq.push({d[v],v});
  }
}
